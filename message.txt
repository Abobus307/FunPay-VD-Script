local function getHighlight(target, color, name)
    if not espCache[target] then espCache[target] = {} end
    local hl = espCache[target][name]
    
    if not hl then
        hl = target:FindFirstChild(name) or Instance.new("Highlight")
        hl.Name = name
        hl.FillColor = color
        hl.FillTransparency = 1
        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
        hl.OutlineTransparency = 1
        hl.Adornee = target
        hl.Parent = target
        espCache[target][name] = hl
    end
    
    if not hl.Enabled or hl.FillTransparency == 1 then
        hl.Enabled = true
        hl.FillColor = color
        TweenService:Create(hl, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {FillTransparency = 0.5, OutlineTransparency = 0}):Play()
    else
        hl.FillColor = color
    end
    
    return hl
end

local function disableHighlight(target, name)
    local hl = nil
    if espCache[target] and espCache[target][name] then
        hl = espCache[target][name]
    else
        hl = target:FindFirstChild(name)
    end
    
    if hl and hl.Enabled and hl.FillTransparency ~= 1 then
        local tween = TweenService:Create(hl, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {FillTransparency = 1, OutlineTransparency = 1})
        tween.Completed:Connect(function()
            if hl.FillTransparency == 1 then
                hl.Enabled = false
            end
        end)
        tween:Play()
    end
end

local function clearAllESP(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then disableHighlight(p.Character, name) end
    end
    for _, obj in ipairs(Generators) do
        if obj.Parent then disableHighlight(obj, name) end
    end
end

local function updateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local isKiller = p:GetAttribute("Killer") or p:GetAttribute("Role") == "Killer" or p.Name == "Killer" or (p.Team and (p.Team.Name == "Killer" or p.TeamColor == BrickColor.new("Really red")))
            if isKiller then
                if Flags.EspKiller then getHighlight(p.Character, Color3.fromRGB(255, 60, 60), "KillerESP") else disableHighlight(p.Character, "KillerESP") end
                disableHighlight(p.Character, "SurvivorESP")
            else
                if Flags.EspSurvivors then getHighlight(p.Character, Color3.fromRGB(80, 170, 255), "SurvivorESP") else disableHighlight(p.Character, "SurvivorESP") end
                disableHighlight(p.Character, "KillerESP")
            end
        end
    end
    for _, obj in ipairs(Generators) do
        if obj.Parent then
            if Flags.EspGenerators then getHighlight(obj, Color3.fromRGB(255, 210, 60), "GeneratorESP") else disableHighlight(obj, "GeneratorESP") end
        end
    end
end

local function PressSkill()
    if IsMobile then
        local btn = PlayerGui:FindFirstChild("check", true) or PlayerGui:FindFirstChild("repair", true) or PlayerGui:FindFirstChild("action", true)
        if btn and btn:IsA("GuiObject") then
            local pos = btn.AbsolutePosition
            local size = btn.AbsoluteSize
            local inset = GuiService:GetGuiInset()
            local x = pos.X + (size.X / 2) + inset.X
            local y = pos.Y + (size.Y / 2) + inset.Y
            pcall(function() VirtualInputManager:SendTouchEvent(8822, Enum.UserInputState.Begin.Value, x, y) end)
            task.wait(0.01)
            pcall(function() VirtualInputManager:SendTouchEvent(8822, Enum.UserInputState.End.Value, x, y) end)
            pcall(function() if firesignal and btn.MouseButton1Click then firesignal(btn.MouseButton1Click) end end)
        end
    else
        pcall(function() VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Space, false, game) end)
        task.wait(0.01)
        pcall(function() VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Space, false, game) end)
    end
end

local function GetSkillCheck()
    for _, guiName in ipairs({"SkillCheckPromptGui", "SkillCheckPromptGui-con"}) do
        local gui = PlayerGui:FindFirstChild(guiName, true)
        if gui then
            local check = gui:FindFirstChild("Check", true)
            if check and check.Visible then
                local line = check:FindFirstChild("Line", true)
                local goal = check:FindFirstChild("Goal", true)
                if line and goal then return line, goal end
            end
        end
    end
    return nil, nil
end

local function AngularDelta(from, to)
    return (to - from + 540) % 360 - 180
end

local function CrossedZone(prevLr, lr, startPos, endPos)
    local function inZone(r)
        if startPos > endPos then return r >= startPos or r <= endPos else return r >= startPos and r <= endPos end
    end
    if inZone(lr) then return true end
    if prevLr == nil then return false end
    
    local moved = (lr - prevLr + 360) % 360
    if moved > 0 and moved < 90 then
        local distToStart = (startPos - prevLr + 360) % 360
        if moved >= distToStart then return true end
    end
    return false
end

local function NormalModeUpdate()
    local line, goal = GetSkillCheck()
    if not (line and goal) then
        lastGoalRotation = nil
        hasClickedThisGoal = false
        lastLineRotation = nil
        lastTick = nil
        wasActive = false
        return
    end
    local lr = line.Rotation % 360
    local gr = goal.Rotation % 360
    local currentTick = os.clock()
    
    if not wasActive then
        wasActive = true
        hasClickedThisGoal = false
        lastGoalRotation = gr
        lastLineRotation = lr
        lastTick = currentTick
        return
    end
    
    if lastGoalRotation ~= nil then
        if math.abs(AngularDelta(lastGoalRotation, gr)) > 5 then
            hasClickedThisGoal = false
            lastLineRotation = nil
            lastTick = nil
        end
    end
    lastGoalRotation = gr
    
    if hasClickedThisGoal then
        lastLineRotation = lr
        lastTick = currentTick
        return
    end
    
    if lastLineRotation and lastTick then
        local dt = currentTick - lastTick
        if dt > 0 then
            local lineSpeed = AngularDelta(lastLineRotation, lr) / dt
            local predictedLr = (lr + lineSpeed * dt * LATENCY_COMPENSATION) % 360
            local startPos = (gr + TARGET_OFFSET - TOLERANCE) % 360
            local endPos = (gr + TARGET_OFFSET + TOLERANCE) % 360
            if CrossedZone(lastLineRotation, predictedLr, startPos, endPos) then
                hasClickedThisGoal = true
                task.spawn(function()
                    task.wait(0.03)
                    PressSkill()
                end)
            end
        end
    end
    lastLineRotation = lr
    lastTick = currentTick
end

local function PerfectModeUpdate()
    local line, goal = GetSkillCheck()
    if not (line and goal) then
        Perfect_LastGoalRotation = nil
        Perfect_HasClickedThisGoal = false
        Perfect_LastLineRotation = nil
        Perfect_LastTick = nil
        Perfect_WasActive = false
        return
    end
    local lr = line.Rotation % 360
    local gr = goal.Rotation % 360
    local currentTick = os.clock()
    
    if not Perfect_WasActive then
        Perfect_WasActive = true
        Perfect_HasClickedThisGoal = false
        Perfect_LastGoalRotation = gr
        Perfect_LastLineRotation = lr
        Perfect_LastTick = currentTick
        return
    end
    
    if Perfect_LastGoalRotation ~= nil then
        if math.abs(AngularDelta(Perfect_LastGoalRotation, gr)) > 5 then
            Perfect_HasClickedThisGoal = false
            Perfect_LastLineRotation = nil
            Perfect_LastTick = nil
        end
    end
    Perfect_LastGoalRotation = gr
    
    if Perfect_HasClickedThisGoal then
        Perfect_LastLineRotation = lr
        Perfect_LastTick = currentTick
        return
    end
    
    if Perfect_LastLineRotation and Perfect_LastTick then
        local dt = currentTick - Perfect_LastTick
        if dt > 0 then
            local lineSpeed = AngularDelta(Perfect_LastLineRotation, lr) / dt
            local predictedLr = (lr + lineSpeed * dt * LATENCY_COMPENSATION) % 360
            local startPos = (gr + 104) % 360
            local endPos = (gr + 108) % 360
            if CrossedZone(Perfect_LastLineRotation, predictedLr, startPos, endPos) then
                Perfect_HasClickedThisGoal = true
                PressSkill()
            end
        end
    end
    Perfect_LastLineRotation = lr
    Perfect_LastTick = currentTick
end

local function InstantModeUpdate()
    local line, goal = GetSkillCheck()
    if not (line and goal) then
        Instant_LastGoalInstance = nil
        Instant_HasClicked = false
        return
    end
    
    if Instant_LastGoalInstance ~= goal then
        Instant_LastGoalInstance = goal
        Instant_HasClicked = false
    end
    
    if not Instant_HasClicked then
        Instant_HasClicked = true
        task.spawn(function()
            task.wait(0.01)
            local currentLine, currentGoal = GetSkillCheck()
            if currentLine and currentGoal then
                local perfectRot = (currentGoal.Rotation % 360 + 106) % 360
                pcall(function() currentLine.Rotation = perfectRot end)
                PressSkill()
            end
        end)
    end
end

local lastSkillCheckTime = 0

RunService.RenderStepped:Connect(function()
    if not Flags.AutoSkillCheck or not ScreenGui.Parent then return end

    if Config.Version == "V1" then
        local gui = PlayerGui:FindFirstChild(SETTINGS.GuiName)
        if gui then
            local mainFrame = gui:FindFirstChild(SETTINGS.MainFrameName)
            if mainFrame and mainFrame.Visible then
                local needle = mainFrame:FindFirstChild(SETTINGS.NeedleName)
                local target = mainFrame:FindFirstChild(SETTINGS.TargetName)
                if needle and target then
                    local needleRotation = needle.Rotation % 360
                    local targetRotation = target.Rotation % 360
                    local zoneStart = (targetRotation + SETTINGS.AngleOffsetStart - 3) % 360
                    local zoneEnd = (targetRotation + SETTINGS.AngleOffsetEnd + 1) % 360
                    local isInsideZone = false
                    if zoneStart > zoneEnd then
                        if needleRotation >= zoneStart or needleRotation <= zoneEnd then isInsideZone = true end
                    else
                        if needleRotation >= zoneStart and needleRotation <= zoneEnd then isInsideZone = true end
                    end
                    if isInsideZone and (os.clock() - lastSkillCheckTime > 0.4) then
                        lastSkillCheckTime = os.clock()
                        PressSkill()
                    end
                end
            end
        end
    elseif Config.Version == "V2" then
        if Config.Method == "Normal" then
            NormalModeUpdate()
        elseif Config.Method == "Perfect" then
            PerfectModeUpdate()
        elseif Config.Method == "Instant" then
            InstantModeUpdate()
        end
    end
end)
